# –––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––– 
# 1 Keyword + Variations

# /(^|[-_/])(cunt(bag|s|whore|y)?)([-_/]|$)/i

# Matches: cunt, cuntbag, cunts, cuntwhore, cunty
# Use this syntax ONLY if the root word is 'unambiguous'

# /(^|[-_/])(ass(es|face|grab|hole))([-_/]|$)/i

# Matches: 
# asses, assface, assgrab, asshole but does NOT match ass
# Omits ? because the root word is ambiguous and may form part of other words, ex. assign

# ––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––  
# 1 Keyword w/t Multiple Characters & Variations

# Basic
# [-_/] = - _ / 

# /(^|[-_/])(poop(s|ing|ed|er(s|z)?|[-_/]?(face|hole|head))?)([-_/]|$)/i

# Matches: 
# poop, poops, pooping, pooped, pooper, poopers, pooperz, poopface, poophole, poophead
# poop-face, poop-head, poop-hole, poop_face, poop_head, poop_hole, poop/face, poop/head, poop/hole

# Advanced
# [-_/.+&=:~,;] = - _ / . + & = : ~ , ;

/(^|[-_/.+&=:~,;])(poop(s|ing|ed|er(s|z)?|[-_/.+&=:~,;]?(face|hole|head))?)([-_/.+&=:~,;]|$)/i

# Matches: 
# Everything above, but with more characters in between
