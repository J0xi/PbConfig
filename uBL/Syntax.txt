/^https?:\/\/[^\/?#]*[0-9]+[-.]?bet[^\/?#]*(?:[\/?#]|$)/i
/^https?:\/\/[^\/?#]*bet[-.]?[0-9]+[^\/?#]*(?:[\/?#]|$)/i

> [0-9]+bet
Term Matches: 1bet, 22bet, 365bet, 777bet
URL Matches: https://1bet.example, https://22bet.com, https://365bet.net, https://my777bet-site.example

> bet[0-9]+
Term Matches: bet1, bet22, bet365, bet777
URL Matches: https://bet1.example, https://bet22.com, https://bet365.net, https://mybet777-site.example

> [-.]?bet
Term Matches: 22-bet, 22.bet, 365-bet, 365.bet
URL Matches: https://22.bet.example,  https://365-bet.example.com

> bet[-.]?
Term Matches: bet-22, bet.22, bet-365, bet.365
URL Matches: https://bet.22.example,  https://bet-365.example.com

# ––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––––  
# 1 Keyword w/t Multiple Characters & Variations

# Basic
# [-_/] = - _ / 

# /(^|[-_/])(poop(s|ing|ed|er(s|z)?|[-_/]?(face|hole|head))?)([-_/]|$)/i

# Matches: 
# poop, poops, pooping, pooped, pooper, poopers, pooperz, poopface, poophole, poophead
# poop-face, poop-head, poop-hole, poop_face, poop_head, poop_hole, poop/face, poop/head, poop/hole

# Advanced
# [-_/.+&=:~,;] = - _ / . + & = : ~ , ;

/(^|[-_/.+&=:~,;])(poop(s|ing|ed|er(s|z)?|[-_/.+&=:~,;]?(face|hole|head))?)([-_/.+&=:~,;]|$)/i

# Matches: 
# Everything above, but with more characters in between
